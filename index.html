// GET /api/ideas -> list all ideas, newest first // POST /api/ideas -> create a new idea, body: { title, body } function rowToIdea(row) { return { id: row.id, number: row.number, title: row.title, body: row.body, status: row.status, createdAt: row.created_at, develop: { problem: row.develop_problem || '', who: row.develop_who || '', how: row.develop_how || '', nextStep: row.develop_next_step || '', notes: row.develop_notes || '' }, aiNotes: JSON.parse(row.ai_notes || '[]') }; } export async function onRequestGet(context) { const { env } = context; try { const { results } = await env.DB.prepare( 'SELECT * FROM ideas ORDER BY created_at DESC' ).all(); return Response.json(results.map(rowToIdea)); } catch (err) { return new Response(JSON.stringify({ error: 'Failed to load ideas', detail: String(err) }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } } export async function onRequestPost(context) { const { env, request } = context; try { const payload = await request.json(); const title = (payload.title || '').toString().slice(0, 500); const body = (payload.body || '').toString(); if (!body.trim()) { return new Response(JSON.stringify({ error: 'Idea body is required' }), { status: 400, headers: { 'Content-Type': 'application/json' } }); } const id = 'idea_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8); const createdAt = Date.now(); const countRow = await env.DB.prepare('SELECT COUNT(*) as c FROM ideas').first(); const number = String((countRow?.c || 0) + 1).padStart(4, '0'); await env.DB.prepare( `INSERT INTO ideas (id, number, title, body, status, created_at, develop_problem, develop_who, develop_how, develop_next_step, develop_notes, ai_notes) VALUES (?, ?, ?, ?, 'spark', ?, '', '', '', '', '', '[]')` ).bind(id, number, title, body, createdAt).run(); return Response.json({ id, number, title, body, status: 'spark', createdAt, develop: { problem: '', who: '', how: '', nextStep: '', notes: '' }, aiNotes: [] }); } catch (err) { return new Response(JSON.stringify({ error: 'Failed to create idea', detail: String(err) }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } }
